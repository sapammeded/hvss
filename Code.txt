const SPREADSHEET_ID="1z8oI95w0JITz9ILxLUjAszg0ijg4qz75huQiwRSkVBA";
const VISITOR_SHEET="Visitor Registration";
const KEY_SHEET="Key Loan";
const VISITOR_HEADERS=["No","Date","Time In","Time Out","Name","Company / PT","ID Card","Purpose","Area","Remark","Status"];
const KEY_HEADERS=["No","Date Out","Time Out","Borrower","Division","Security","Key Name","Key Number","Date In","Time In","Returner","Remark OUT","Status"];

function doGet(e){
  try{
    const action=(e&&e.parameter&&e.parameter.action)||"health";
    if(action==="getAll") return json_(getAll_());
    if(action==="setupCentral") return json_(setup_());
    if(action==="health") return json_({ok:true,service:"HVSS2",spreadsheetId:SPREADSHEET_ID});
    return json_({ok:true,service:"HVSS2",spreadsheetId:SPREADSHEET_ID});
  }catch(err){return json_({ok:false,error:String(err)});}
}

function doPost(e){
  try{
    const raw=(e&&e.postData&&e.postData.contents)||"{}";
    const p=JSON.parse(raw);
    if(p.action==="setupCentral") return json_(setup_());
    if(p.action==="syncAll") return json_(syncAll_(p));
    if(p.action==="addVisitor") return json_(addVisitor_(p.visitor||{}));
    if(p.action==="addKey") return json_(addKey_(p.key||{}));
    return json_({ok:false,error:"Unknown action: "+p.action});
  }catch(err){return json_({ok:false,error:String(err),stack:err.stack||""});}
}

function ss_(){return SpreadsheetApp.openById(SPREADSHEET_ID);}

function ensure_(ss,name,headers){
  let sh=ss.getSheetByName(name);
  if(!sh) sh=ss.insertSheet(name);
  const current=sh.getRange(1,1,1,headers.length).getValues()[0];
  let mismatch=false;
  for(let i=0;i<headers.length;i++) if(String(current[i]||"")!==headers[i]) {mismatch=true;break;}
  if(mismatch) sh.getRange(1,1,1,headers.length).setValues([headers]);
  sh.setFrozenRows(1);
  return sh;
}

function setup_(){
  const ss=ss_();
  const v=ensure_(ss,VISITOR_SHEET,VISITOR_HEADERS);
  const k=ensure_(ss,KEY_SHEET,KEY_HEADERS);
  return {ok:true,spreadsheetId:SPREADSHEET_ID,visitorSheet:v.getName(),keySheet:k.getName()};
}

function addVisitor_(x){
  const sh=ensure_(ss_(),VISITOR_SHEET,VISITOR_HEADERS);
  const no=Math.max(1,sh.getLastRow());
  sh.appendRow([no,x.date||"",x.timeIn||"",x.timeOut||"",x.name||"",x.company||"",x.idcard||"",x.purpose||"",x.area||"",x.remark||"",x.timeOut?"OUT":"INSIDE"]);
  SpreadsheetApp.flush();
  return {ok:true,row:sh.getLastRow(),visitorCount:Math.max(0,sh.getLastRow()-1)};
}

function addKey_(x){
  const sh=ensure_(ss_(),KEY_SHEET,KEY_HEADERS);
  const no=Math.max(1,sh.getLastRow());
  sh.appendRow([no,x.outDate||"",x.outTime||"",x.borrower||"",x.division||"",x.security||"",x.keyName||"",x.keyNumber||"",x.inDate||"",x.inTime||"",x.returner||"",x.remark||"",x.inDate?"RETURNED":"KEY OUT"]);
  SpreadsheetApp.flush();
  return {ok:true,row:sh.getLastRow(),keyCount:Math.max(0,sh.getLastRow()-1)};
}

function syncAll_(p){
  const lock=LockService.getScriptLock();
  lock.waitLock(30000);
  try{
    const ss=ss_();
    const v=ensure_(ss,VISITOR_SHEET,VISITOR_HEADERS);
    const k=ensure_(ss,KEY_SHEET,KEY_HEADERS);
    const vs=Array.isArray(p.visitors)?p.visitors:[];
    const ks=Array.isArray(p.keys)?p.keys:[];
    let visitorWritten=0,keyWritten=0;

    // IMPORTANT: never touch spreadsheet column widths or other formatting.
    // Only replace the data rows for datasets actually supplied by the client.
    if(vs.length){
      if(v.getLastRow()>1) v.getRange(2,1,v.getLastRow()-1,VISITOR_HEADERS.length).clearContent();
      v.getRange(2,1,vs.length,VISITOR_HEADERS.length).setValues(vs.map((x,i)=>[
        i+1,x.date||"",x.timeIn||"",x.timeOut||"",x.name||"",x.company||"",x.idcard||"",x.purpose||"",x.area||"",x.remark||"",x.timeOut?"OUT":"INSIDE"
      ]));
      visitorWritten=vs.length;
    }
    if(ks.length){
      if(k.getLastRow()>1) k.getRange(2,1,k.getLastRow()-1,KEY_HEADERS.length).clearContent();
      k.getRange(2,1,ks.length,KEY_HEADERS.length).setValues(ks.map((x,i)=>[
        i+1,x.outDate||"",x.outTime||"",x.borrower||"",x.division||"",x.security||"",x.keyName||"",x.keyNumber||"",x.inDate||"",x.inTime||"",x.returner||"",x.remark||"",x.inDate?"RETURNED":"KEY OUT"
      ]));
      keyWritten=ks.length;
    }
    SpreadsheetApp.flush();
    return {ok:true,visitorCount:visitorWritten,keyCount:keyWritten,visitorSheet:VISITOR_SHEET,keySheet:KEY_SHEET};
  }finally{lock.releaseLock();}
}

function getAll_(){
  const ss=ss_();
  const v=ensure_(ss,VISITOR_SHEET,VISITOR_HEADERS);
  const k=ensure_(ss,KEY_SHEET,KEY_HEADERS);
  return {ok:true,visitors:rows_(v,VISITOR_HEADERS),keys:rows_(k,KEY_HEADERS)};
}

function rows_(sh,headers){
  const n=sh.getLastRow();
  if(n<2)return [];
  const values=sh.getRange(2,1,n-1,headers.length).getValues();
  return values.map(function(r,i){
    if(headers===VISITOR_HEADERS){
      return {id:"sheet-v-"+(i+2),date:r[1],timeIn:r[2],timeOut:r[3],name:r[4],company:r[5],idcard:r[6],purpose:r[7],area:r[8],remark:r[9],status:r[10]};
    }
    return {id:"sheet-k-"+(i+2),outDate:r[1],outTime:r[2],borrower:r[3],division:r[4],security:r[5],keyName:r[6],keyNumber:r[7],inDate:r[8],inTime:r[9],returner:r[10],remark:r[11],status:r[12]};
  });
}

function json_(o){
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
