const SPREADSHEET_ID = '1z8oI95w0JITz9ILxLUjAszg0ijg4qz75huQiwRSkVBA';
const VISITOR_SHEET = 'Visitor Registration';
const KEY_SHEET = 'Key Loan';
const VISITOR_HEADERS = ['No','Date','Time In','Time Out','Name','Company / PT','ID Card','Purpose','Area','Remark','Status'];
const KEY_HEADERS = ['No','Date Out','Time Out','Nama Peminjam','Divisi','Security OUT','Nama Kunci','Key Number','Date In','Time In','Nama Yang Mengembalikan','Divisi IN','Security IN','Remark OUT','Keterangan','Status'];

function json_(o){return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON)}
function jsonp_(o,cb){return ContentService.createTextOutput(cb+'('+JSON.stringify(o)+')').setMimeType(ContentService.MimeType.JAVASCRIPT)}
function ss_(){return SpreadsheetApp.openById(SPREADSHEET_ID)}

function normId_(v){
  // ID Card is an identifier, NEVER a number. Preserve exactly what the user typed, including leading zeroes.
  return String(v==null?'':v).trim();
}
function normTime_(v){
  v=String(v==null?'':v).trim();
  if(!v)return '';
  const m=v.match(/^(\d{1,2}):(\d{1,2})/);
  return m ? String(parseInt(m[1],10)).padStart(2,'0')+':'+String(parseInt(m[2],10)).padStart(2,'0') : v;
}
function dateKey_(v){
  v=String(v==null?'':v).trim(); if(!v)return '';
  let m=v.match(/^(\d{4})-(\d{2})-(\d{2})$/); if(m)return v;
  m=v.match(/^(?:[^,]+,\s*)?(\d{1,2})[-\/](\d{1,2})[-\/](\d{4})$/);
  return m?m[3]+'-'+String(m[2]).padStart(2,'0')+'-'+String(m[1]).padStart(2,'0'):v;
}
function dateLabel_(v){
  const k=dateKey_(v); if(!k)return '';
  const p=k.split('-').map(Number),dt=new Date(p[0],p[1]-1,p[2]);
  const days=['Minggu','Senin','Selasa','Rabu','Kamis','Jumat','Sabtu'];
  return days[dt.getDay()]+', '+String(p[2]).padStart(2,'0')+'-'+String(p[1]).padStart(2,'0')+'-'+p[0];
}
function ensureSheet_(name,headers){
  const ss=ss_(); let sh=ss.getSheetByName(name); if(!sh) sh=ss.insertSheet(name);
  sh.getRange(1,1,1,headers.length).setValues([headers]);
  return sh;
}
function prepareTextColumns_(sh,name){
  // ALL database columns are TEXT. This prevents Google Sheets from converting
  // identifiers such as 001, 0076, 0008, etc. into numbers and dropping zeroes.
  // Dates/times are intentionally stored as display strings too, so 08:34 stays 08:34.
  const rows=Math.max(sh.getMaxRows()-1,1);
  sh.getRange(2,1,rows,headers.length).setNumberFormat('@');
}
function readSheet_(name,headers){
  const sh=ensureSheet_(name,headers), lr=sh.getLastRow();
  if(lr<2)return [];
  const vals=sh.getRange(2,1,lr-1,headers.length).getDisplayValues();
  const notes=sh.getRange(2,1,lr-1,1).getNotes();
  return vals.map((r,i)=>rowToObj_(name,r,notes[i]&&notes[i][0])).filter(r=>r.__hasData);
}
function rowToObj_(name,r,noteId){
  const has=r.some(v=>String(v).trim()!=='');
  const legacyId=String(r[0]||'').trim();
  const id=String(noteId||'').trim() || (/^\d+$/.test(legacyId)?Utilities.getUuid():legacyId) || Utilities.getUuid();
  if(name===VISITOR_SHEET)return {id,date:r[1]||'',timeIn:normTime_(r[2]),timeOut:normTime_(r[3]),name:r[4]||'',company:r[5]||'',idcard:normId_(r[6]),purpose:r[7]||'',area:r[8]||'',remark:r[9]||'',status:r[10]||'',__hasData:has};
  return {id,outDate:r[1]||'',outTime:normTime_(r[2]),borrower:r[3]||'',division:r[4]||'',security:r[5]||'',keyName:r[6]||'',keyNumber:r[7]||'',inDate:r[8]||'',inTime:normTime_(r[9]),returner:r[10]||'',inDivision:r[11]||'',inSecurity:r[12]||'',remark:r[13]||'',note:r[14]||'',status:r[15]||'',__hasData:has};
}
function objToRow_(name,x){
  if(name===VISITOR_SHEET)return [x.id||Utilities.getUuid(),dateLabel_(x.date),normTime_(x.timeIn),normTime_(x.timeOut),x.name||'',x.company||'',normId_(x.idcard),x.purpose||'',x.area||'',x.remark||'',x.timeOut?'OUT':'INSIDE'];
  return [x.id||Utilities.getUuid(),dateLabel_(x.outDate),normTime_(x.outTime),x.borrower||'',x.division||'',x.security||'',x.keyName||'',x.keyNumber==null?'':String(x.keyNumber),dateLabel_(x.inDate),normTime_(x.inTime),x.returner||'',x.inDivision||'',x.inSecurity||'',x.remark||'',x.note||'',x.inDate?'RETURNED':'KEY OUT'];
}
function clean_(x){const y={...x};delete y.__hasData;return y}
function merge_(existing,incoming){
  const map=new Map(existing.map(x=>[String(x.id),clean_(x)]));
  incoming.forEach(raw=>{const x=clean_(raw),id=String(x.id||Utilities.getUuid());x.id=id;map.set(id,x)});
  return Array.from(map.values());
}
function writeMerged_(name,headers,incoming){
  const sh=ensureSheet_(name,headers), existing=readSheet_(name,headers), merged=merge_(existing,incoming);
  prepareTextColumns_(sh,name);
  const oldRows=Math.max(0,sh.getMaxRows()-1);
  if(oldRows>0){sh.getRange(2,1,oldRows,headers.length).clearContent();sh.getRange(2,1,oldRows,1).clearNote();}
  if(merged.length){
    const rows=merged.map(x=>objToRow_(name,x));
    sh.getRange(2,1,rows.length,headers.length).setValues(rows);
    // Visible No is always 1,2,3...; internal UUID is kept invisibly as a note on column A.
    sh.getRange(2,1,rows.length,1).setValues(merged.map((_,i)=>[i+1]));
    sh.getRange(2,1,rows.length,1).setNotes(merged.map(x=>[String(x.id||Utilities.getUuid())]));
  }
  return merged.length;
}
function doGet(e){
  try{
    const a=(e.parameter.action||'').toLowerCase(),cb=e.parameter.callback;
    if(a==='getcentral'){
      const out={ok:true,visitors:readSheet_(VISITOR_SHEET,VISITOR_HEADERS),keys:readSheet_(KEY_SHEET,KEY_HEADERS),serverTime:new Date().toISOString()};
      return cb?jsonp_(out,cb):json_(out);
    }
    return json_({ok:true,service:'HVSS CENTRAL',spreadsheetId:SPREADSHEET_ID});
  }catch(err){const out={ok:false,error:String(err&&err.message||err)};return e.parameter.callback?jsonp_(out,e.parameter.callback):json_(out)}
}
function doPost(e){
  try{
    const p=JSON.parse(e.postData.contents||'{}'),a=(p.action||'').toLowerCase();
    if(p.spreadsheetId && p.spreadsheetId!==SPREADSHEET_ID)throw new Error('Spreadsheet ID tidak cocok');
    if(a==='syncall'){
      const v=writeMerged_(VISITOR_SHEET,VISITOR_HEADERS,Array.isArray(p.visitors)?p.visitors:[]);
      const k=writeMerged_(KEY_SHEET,KEY_HEADERS,Array.isArray(p.keys)?p.keys:[]);
      return json_({ok:true,visitorCount:v,keyCount:k,serverTime:new Date().toISOString(),message:'CENTRAL VERIFIED'});
    }
    if(a==='resetcentral'){
      [[VISITOR_SHEET,VISITOR_HEADERS],[KEY_SHEET,KEY_HEADERS]].forEach(([n,h])=>{const sh=ensureSheet_(n,h);prepareTextColumns_(sh,n);if(sh.getMaxRows()>1){sh.getRange(2,1,sh.getMaxRows()-1,h.length).clearContent();sh.getRange(2,1,sh.getMaxRows()-1,1).clearNote();}});
      return json_({ok:true,visitorCount:0,keyCount:0,message:'CENTRAL RESET'});
    }
    return json_({ok:false,error:'Unknown action'});
  }catch(err){return json_({ok:false,error:String(err&&err.message||err)})}
}
