const SPREADSHEET_ID = "1z8oI95w0JITz9ILxLUjAszg0ijg4qz75huQiwRSkVBA";
const VISITOR_SHEET = "Visitor Registration";
const KEY_SHEET = "Key Loan";

const VISITOR_HEADERS = ["No", "Date", "Time In", "Time Out", "Name", "Company / PT", "ID Card", "Purpose", "Area", "Remark", "Status"];
const KEY_HEADERS = ["No", "Date Out", "Time Out", "Borrower", "Division", "Security", "Key Name", "Key Number", "Date In", "Time In", "Returner", "Status"];

function doGet() {
  return ContentService.createTextOutput(JSON.stringify({ok:true,service:"HVSS",spreadsheetId:SPREADSHEET_ID}))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const p = JSON.parse(e.postData.contents || "{}");
    if (p.action === "setupCentral") return json_(setupCentral_());
    if (p.action === "syncAll") return json_(syncAll_(p));
    return json_({ok:false,error:"Unknown action"});
  } catch(err) {
    return json_({ok:false,error:String(err)});
  }
}

function setupCentral_() {
  const ss=SpreadsheetApp.openById(SPREADSHEET_ID);
  const v=getOrCreate_(ss,VISITOR_SHEET,VISITOR_HEADERS);
  const k=getOrCreate_(ss,KEY_SHEET,KEY_HEADERS);
  return {ok:true,visitorSheet:v.getName(),keySheet:k.getName(),spreadsheetId:SPREADSHEET_ID};
}

function getOrCreate_(ss,name,headers) {
  let sh=ss.getSheetByName(name);
  if(!sh) sh=ss.insertSheet(name);
  sh.clearContents();
  sh.getRange(1,1,1,headers.length).setValues([headers]);
  sh.setFrozenRows(1);
  return sh;
}

function syncAll_(p) {
  const ss=SpreadsheetApp.openById(SPREADSHEET_ID);
  const v=getOrCreate_(ss,VISITOR_SHEET,VISITOR_HEADERS);
  const k=getOrCreate_(ss,KEY_SHEET,KEY_HEADERS);
  const visitors=Array.isArray(p.visitors)?p.visitors:[];
  const keys=Array.isArray(p.keys)?p.keys:[];
  if(visitors.length) v.getRange(2,1,visitors.length,VISITOR_HEADERS.length).setValues(
    visitors.map((x,i)=>[i+1,x.date||"",x.timeIn||"",x.timeOut||"",x.name||"",x.company||"",x.idcard||"",x.purpose||"",x.area||"",x.remark||"",x.timeOut?"OUT":"INSIDE"])
  );
  if(keys.length) k.getRange(2,1,keys.length,KEY_HEADERS.length).setValues(
    keys.map((x,i)=>[i+1,x.outDate||"",x.outTime||"",x.borrower||"",x.division||"",x.security||"",x.keyName||"",x.keyNumber||"",x.inDate||"",x.inTime||"",x.returner||"",x.inDate?"RETURNED":"KEY OUT"])
  );
  return {ok:true,visitorCount:visitors.length,keyCount:keys.length};
}

function json_(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}
